cmake_minimum_required(VERSION 3.16)
project(flir_eye_tracker_record LANGUAGES CXX)

find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
find_package(RapidJSON CONFIG REQUIRED)
pkg_check_modules(ZMQ REQUIRED IMPORTED_TARGET libzmq)

add_library(eye_tracker_control src/eye_trackers.cpp src/clock_measurements.cpp)
target_compile_features(eye_tracker_control PUBLIC cxx_std_17)
target_include_directories(eye_tracker_control PUBLIC src ${RAPIDJSON_INCLUDE_DIRS})
target_link_libraries(eye_tracker_control PUBLIC PkgConfig::ZMQ Threads::Threads)
target_compile_options(eye_tracker_control PRIVATE -Wall -Wextra -Wpedantic)

option(BUILD_FLIR_EYE_RECORDER "Build the FLIR recording executable" ON)
if(BUILD_FLIR_EYE_RECORDER)
  find_package(fmt CONFIG REQUIRED)
  if(NOT TARGET flirmulticamera)
    find_package(cpp_utils CONFIG REQUIRED)
    find_package(flirmulticamera CONFIG REQUIRED)
  endif()
  find_package(spdlog REQUIRED)
  add_executable(flir_eye_tracker_recorder src/record.cpp)
  target_link_libraries(flir_eye_tracker_recorder PRIVATE
    eye_tracker_control flirmulticamera spdlog::spdlog)
  target_compile_options(flir_eye_tracker_recorder PRIVATE -Wall -Wextra -Wpedantic)
  install(TARGETS flir_eye_tracker_recorder RUNTIME DESTINATION bin)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(eye_tracker_control_test tests/eye_trackers_test.cpp)
  target_link_libraries(eye_tracker_control_test PRIVATE eye_tracker_control)
  add_test(NAME eye_tracker_control COMMAND eye_tracker_control_test)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME align_pupil_timestamps COMMAND ${Python3_EXECUTABLE}
      "${CMAKE_CURRENT_SOURCE_DIR}/tests/align_pupil_timestamps_test.py")
  endif()
endif()

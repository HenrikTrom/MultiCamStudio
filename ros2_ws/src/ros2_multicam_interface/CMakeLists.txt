cmake_minimum_required(VERSION 3.16)
project(ros2_multicam_interface)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(spdlog REQUIRED)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET
  libavcodec libavformat libavutil libswscale)

add_executable(camera_tab_ros2_web_streamer src/camera_tab_ros2_web_streamer.cpp)
target_compile_features(camera_tab_ros2_web_streamer PRIVATE cxx_std_17)
target_compile_options(camera_tab_ros2_web_streamer PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(camera_tab_ros2_web_streamer rclcpp sensor_msgs)
target_link_libraries(camera_tab_ros2_web_streamer
  PkgConfig::FFMPEG nlohmann_json::nlohmann_json spdlog::spdlog Threads::Threads)
install(TARGETS camera_tab_ros2_web_streamer DESTINATION lib/${PROJECT_NAME})
find_package(OpenCV 4.7 REQUIRED COMPONENTS core imgproc imgcodecs)
add_library(mcs_synchronized_recorder STATIC src/synchronized_recorder.cpp)
target_compile_features(mcs_synchronized_recorder PUBLIC cxx_std_17)
target_compile_options(mcs_synchronized_recorder PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(mcs_synchronized_recorder PUBLIC include)
ament_target_dependencies(mcs_synchronized_recorder rclcpp sensor_msgs)
target_link_libraries(mcs_synchronized_recorder nlohmann_json::nlohmann_json Threads::Threads)
foreach(recorder calibration_tab_ros2_image_recorder)
  add_executable(${recorder} src/${recorder}.cpp)
  target_link_libraries(${recorder} mcs_synchronized_recorder)
  install(TARGETS ${recorder} DESTINATION lib/${PROJECT_NAME})
endforeach()
target_include_directories(calibration_tab_ros2_image_recorder PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(calibration_tab_ros2_image_recorder ${OpenCV_LIBS})
install(DIRECTORY scripts/ DESTINATION lib/${PROJECT_NAME} USE_SOURCE_PERMISSIONS)
ament_package()

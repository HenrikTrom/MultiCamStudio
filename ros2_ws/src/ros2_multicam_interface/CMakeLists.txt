cmake_minimum_required(VERSION 3.16)
project(ros2_multicam_interface)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
find_package(spdlog REQUIRED)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FFMPEG REQUIRED IMPORTED_TARGET
  libavcodec libavformat libavutil libswscale)

add_executable(camera_tab_ros2_web_streamer src/camera_tab_ros2_web_streamer.cpp)
target_compile_features(camera_tab_ros2_web_streamer PRIVATE cxx_std_17)
target_compile_options(camera_tab_ros2_web_streamer PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(camera_tab_ros2_web_streamer rclcpp sensor_msgs)
target_link_libraries(camera_tab_ros2_web_streamer
  PkgConfig::FFMPEG nlohmann_json::nlohmann_json spdlog::spdlog Threads::Threads)
install(TARGETS camera_tab_ros2_web_streamer DESTINATION lib/${PROJECT_NAME})
find_package(OpenCV 4.7 REQUIRED COMPONENTS core imgproc imgcodecs)
add_executable(calibration_tab_ros2_image_recorder src/calibration_tab_ros2_image_recorder.cpp)
target_compile_features(calibration_tab_ros2_image_recorder PRIVATE cxx_std_17)
target_compile_options(calibration_tab_ros2_image_recorder PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(calibration_tab_ros2_image_recorder PRIVATE include ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(calibration_tab_ros2_image_recorder rclcpp sensor_msgs)
target_link_libraries(calibration_tab_ros2_image_recorder
  nlohmann_json::nlohmann_json Threads::Threads ${OpenCV_LIBS})
install(TARGETS calibration_tab_ros2_image_recorder DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY scripts/ DESTINATION lib/${PROJECT_NAME} USE_SOURCE_PERMISSIONS)
ament_package()
